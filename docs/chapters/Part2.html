import folium
from folium.plugins import MarkerCluster

df_map = (
    raw.dropna(subset=["latitude", "longitude"])
       .drop_duplicates(subset="station_code")
       .copy() )
print("Nombre de stations :", len(df_map))

m = folium.Map(
    location=[df_map["latitude"].mean(), df_map["longitude"].mean()],
    zoom_start=11 )
cluster = MarkerCluster().add_to(m)

for _, r in df_map.iterrows():
    folium.Marker(
        location=[r["latitude"], r["longitude"]],
        popup=f"{r['name']}<br>Capacité : {r['capacity']}",).add_to(cluster)

3 Nombre de stations par commune

# Une ligne par station, puis comptage par commune
stations = raw.dropna(subset=["commune"]).drop_duplicates(subset="station_code")
counts = (stations["commune"].value_counts()
          .rename_axis("commune").reset_index(name="n_stations"))

fig = px.bar(
    counts.sort_values("n_stations"),
    x="n_stations", y="commune", orientation="h",
    color_discrete_sequence=["#22C55E"],
    labels={"n_stations": "Nombre de stations", "commune": ""},
    title="Nombre de stations par commune",
    height=max(400, 22 * len(counts)), )
fig.show()

4 Nettoyage et agrégation des données

import numpy as np
import pandas as pd
import plotly.express as px

JOURS = ["Lundi", "Mardi", "Mercredi", "Jeudi", "Vendredi", "Samedi", "Dimanche"]
VERT = ["#DCFCE7", "#86EFAC", "#22C55E", "#15803D", "#14532D"]

TIME_COL = "collected_at_utc"

df = raw.copy()
df["ts"] = pd.to_datetime(df[TIME_COL], errors="coerce", utc=True).dt.tz_convert("Europe/Paris")
df = df.dropna(subset=["ts", "latitude", "longitude"])
df = df[(df["capacity"] > 0) & (df["is_installed"] == 1)].copy()

df["hour"] = df["ts"].dt.hour
df["weekday"] = df["ts"].dt.dayofweek
df["periode"] = np.where(df["weekday"] >= 5, "Week-end", "Semaine")
df["occupancy"] = df["bikes_available"] / df["capacity"]

stations = (
    df.groupby("station_code", as_index=False)
      .agg(name=("name", "first"), commune=("commune", "first"),
           capacity=("capacity", "first"),
           latitude=("latitude", "first"), longitude=("longitude", "first"),
           bikes_available=("bikes_available", "mean"),
           occupancy=("occupancy", "mean"))
)

print("Nombre de stations :", len(stations))
print("Période :", df["ts"].min(), "->", df["ts"].max())
print("Nombre de jours :", df["ts"].dt.date.nunique())

pivot = (df.groupby(["weekday", "hour"])["occupancy"].mean()
           .unstack("hour").reindex(range(7)))

by_commune = (stations.groupby("commune", as_index=False)
                      .agg(n_stations=("station_code", "count"),

                      .sort_values("occupancy"))

fig = px.bar(by_commune, x="occupancy", y="commune", orientation="h",
             color="n_stations", color_continuous_scale=VERT,

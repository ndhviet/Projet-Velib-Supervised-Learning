3 Nombre de stations par commune

Ce diagramme présente le nombre de stations Vélib’ dans chaque commune. Il permet de comparer leur répartition géographique et d’identifier les communes disposant du plus grand nombre de stations.

Code 
# Une ligne par station, puis comptage par commune
stations = raw.dropna(subset=["commune"]).drop_duplicates(subset="station_code")
counts = (stations["commune"].value_counts()
          .rename_axis("commune").reset_index(name="n_stations"))

fig = px.bar(
    counts.sort_values("n_stations"),
    x="n_stations", y="commune", orientation="h",
    color_discrete_sequence=["#22C55E"],
    labels={"n_stations": "Nombre de stations", "commune": ""},
    title="Nombre de stations par commune",
    height=max(400, 22 * len(counts)), )
fig.show()

4 Nettoyage et agrégation des données

Les données sont nettoyées afin de supprimer les observations incomplètes ou invalides. Elles sont ensuite agrégées par station pour calculer des indicateurs moyens, notamment le nombre de vélos disponibles et le taux d’occupation.

Code 
TIME_COL = "collected_at_utc"

df = raw.copy()
df["ts"] = pd.to_datetime(df[TIME_COL], errors="coerce", utc=True).dt.tz_convert("Europe/Paris")
df = df.dropna(subset=["ts", "latitude", "longitude"])
df = df[(df["capacity"] > 0) & (df["is_installed"] == 1)].copy()

df["hour"] = df["ts"].dt.hour
df["weekday"] = df["ts"].dt.dayofweek
df["periode"] = np.where(df["weekday"] >= 5, "Week-end", "Semaine")
df["occupancy"] = df["bikes_available"] / df["capacity"]

stations = (
    df.groupby("station_code", as_index=False)
      .agg(name=("name", "first"), commune=("commune", "first"),
           capacity=("capacity", "first"),
           latitude=("latitude", "first"), longitude=("longitude", "first"),
           bikes_available=("bikes_available", "mean"),
           occupancy=("occupancy", "mean"))
)

print("Nombre de stations :", len(stations))
print("Période :", df["ts"].min(), "->", df["ts"].max())
print("Nombre de jours :", df["ts"].dt.date.nunique())

5.2 Répartition du taux de disponibilité

Code 
import plotly.express as px
import pandas as pd

# Calcul du nombre moyen de vélos par station
dispo = (
    raw.groupby("station_code", as_index=False)
    .agg(
        bikes_available=("bikes_available", "mean"),
        capacity=("capacity", "first")
    )
)

# Éviter les divisions par zéro
dispo = dispo[dispo["capacity"] > 0].copy()

# Taux de disponibilité en pourcentage
dispo["taux_dispo"] = (
    dispo["bikes_available"] / dispo["capacity"] * 100
)

# Création des catégories
bins = [0, 20, 40, 60, 80, 100]
labels = ["0–20%", "20–40%", "40–60%",
          "60–80%", "80–100%"]

dispo["categorie"] = pd.cut(
    dispo["taux_dispo"].clip(0, 100),
    bins=bins,
    labels=labels,
    include_lowest=True
)

counts = (
    dispo["categorie"]
    .value_counts()
    .reindex(labels, fill_value=0)
    .reset_index()
)

counts.columns = ["Taux de disponibilité", "Nombre de stations"]

# Création du graphique
fig = px.bar(
    counts,
    x="Taux de disponibilité",
    y="Nombre de stations",
    color="Taux de disponibilité",
    color_discrete_sequence=VERT,
    text="Nombre de stations",
    title="Répartition des stations selon le taux de disponibilité"
)

fig.update_traces(
    textposition="outside",
    marker_line_width=0
)

fig.update_layout(
    template="plotly_white",
    height=450,
    showlegend=False,
    xaxis_title="Taux moyen de disponibilité",
    yaxis_title="Nombre de stations"
)

fig.show()
